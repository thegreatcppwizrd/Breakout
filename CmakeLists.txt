cmake_minimum_required(VERSION 3.13)
project(Breakout LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

# Force CMake to generate the compilation database that IntelliSense engines read
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)

set(CMAKE_MSVC_RUNTIME_LIBRARY
    "MultiThreaded$<$<CONFIG:Debug>:Debug>")

add_subdirectory("thirdParty/raylib-6.2")

# FIX: Glob both .cpp AND header files so the IDE maps IntelliSense to them
file(GLOB_RECURSE MY_SOURCES CONFIGURE_DEPENDS
    "${CMAKE_CURRENT_SOURCE_DIR}/src/*.cpp"
    "${CMAKE_CURRENT_SOURCE_DIR}/include/headers/*.h"
    "${CMAKE_CURRENT_SOURCE_DIR}/include/headers/*.hpp"
)

add_executable(Breakout ${MY_SOURCES})

target_compile_definitions(Breakout PUBLIC
    RESOURCES_PATH="${CMAKE_CURRENT_SOURCE_DIR}/resources/")

target_include_directories(Breakout PUBLIC
    "${CMAKE_CURRENT_SOURCE_DIR}/include") # Note: Changed to /include so you can write #include "headers/file.h"

target_link_libraries(Breakout PUBLIC raylib)
