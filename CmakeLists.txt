cmake_minimum_required(VERSION 3.13)
project(Breakout LANGUAGES C CXX) 

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

set(CMAKE_EXPORT_COMPILE_COMMANDS ON)

set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")

add_subdirectory("thirdParty/raylib-6.2")

# Automatically scan and glob your source arrays
file(GLOB_RECURSE MY_SOURCES CONFIGURE_DEPENDS
    "${CMAKE_CURRENT_SOURCE_DIR}/src/*.cpp"
    "${CMAKE_CURRENT_SOURCE_DIR}/include/headers/*.h"
    "${CMAKE_CURRENT_SOURCE_DIR}/include/headers/*.hpp"
)

add_executable(Breakout ${MY_SOURCES})

# FIXED: Wrapped the macro definition properly inside safe outer quotes
target_compile_definitions(Breakout PUBLIC
   "RESOURCES_PATH=\"${CMAKE_CURRENT_SOURCE_DIR}/resources/\""
)

target_include_directories(Breakout PUBLIC
    "${CMAKE_CURRENT_SOURCE_DIR}/include"
) 

target_link_libraries(Breakout PUBLIC raylib)
